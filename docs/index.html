8 controles. Uma API.

Showcase.client.lua 
-- Axiom showcases como WindUI, mas com workspace
local Axiom = require(game.ReplicatedStorage.Axiom)
local Window = Axiom:CreateWindow({
  Title="Axiom Control Center", Theme="Dark", Acrylic=true, Blur=true
})
local Tab = Window:AddTab({Name="Dashboard", Icon="home"})
local Main, Inspector = Tab:AddColumnGroup({Ratio=0.62})

Main:AddToggle({Name="Enable Feature", Default=false})
Main:AddSlider({Name="Performance", Min=0, Max=100, Suffix="%"})
Main:AddDropdown({Name="Mode", Options={"Balanced","Performance"}})
Main:AddKeybind({Name="Toggle UI", Default=Enum.KeyCode.RightShift})

Inspector:AddColorPicker({Name="Accent", Default=Color3.fromRGB(139,55,255)})
Axiom:Notify({Title="Welcome to Axiom", Description="Workspace ready."})

  Primary = Color3.fromRGB(0,220,255),
  Secondary = Color3.fromRGB(174,72,255),
  Radius = UDim.new(0,12)
})
Axiom:SetTheme(Neon)
