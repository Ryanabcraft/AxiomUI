-- 1 LINHA — igual Rayfield/WindUI
local Axiom = loadstring(game:HttpGet("https://raw.githubusercontent.com/Ryanabcraft/AxiomUI/main/dist/Axiom.lua"))()

local Window = Axiom:CreateWindow({Title="Meu Hub", Theme="Dark", Acrylic=true, Blur=true})
local Tab = Window:AddTab({Name="Main", Icon="home"})

Tab:AddToggle({Name="Auto Farm", Default=false, Callback=function(v) _G.Farm=v end})
Tab:AddSlider({Name="Speed", Min=16, Max=200, Suffix="%"})
Tab:AddDropdown({Name="Mode", Options={"Legit","Rage"}})
Tab:AddKeybind({Name="Toggle UI", Default=Enum.KeyCode.RightShift})

Axiom:Notify({Title="Carregado!", Description="Axiom pronta."})
